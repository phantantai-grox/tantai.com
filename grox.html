const express = require('express'); const multer = require('multer'); const cors = require('cors'); const fs = require('fs'); const path = require('path'); const app = express(); app.use(cors()); app.use(express.json()); // Thư mục lưu file const uploadDir = path.join(__dirname, 'uploads'); if (!fs.existsSync(uploadDir)) { fs.mkdirSync(uploadDir); } const storage = multer.diskStorage({ destination: (req, file, cb) => cb(null, uploadDir), filename: (req, file, cb) => cb(null, Date.now() + '-' + file.originalname) }); const upload = multer({ storage }); // Database lưu trong file JSON đơn giản const dbPath = path.join(__dirname, 'files_db.json'); function readDB() { if (!fs.existsSync(dbPath)) return {}; try { return JSON.parse(fs.readFileSync(dbPath)); } catch (e) { return {}; } } function writeDB(data) { fs.writeFileSync(dbPath, JSON.stringify(data, null, 2)); } // CẤU HÌNH PHIÊN BẢN & BẢO TRÌ (Chỉnh sửa ở đây khi muốn khóa web) const CONFIG = { LATEST_VERSION: "1.0.0", // Phiên bản mới nhất IS_LOCK: false, // Đặt true nếu muốn khóa toàn bộ hệ thống LOCK_REASON: "Hệ thống đang bảo trì hoặc đã có bản cập nhật mới. Vui lòng liên hệ Admin!" }; // 1. API Kiểm tra phiên bản / Khóa web app.get('/api/check-version', (req, res) => { const clientVersion = req.query.v; if (CONFIG.IS_LOCK) { return res.json({ status: 'LOCKED', reason: CONFIG.LOCK_REASON }); } if (clientVersion !== CONFIG.LATEST_VERSION) { return res.json({ status: 'UPDATE_REQUIRED', reason: `Web của bạn đang ở bản v${clientVersion}. Đã có bản cập nhật v${CONFIG.LATEST_VERSION}! Vui lòng cập nhật.` }); } res.json({ status: 'OK' }); }); // 2. API Upload file từ Admin app.post('/api/upload', upload.single('file'), (req, res) => { const { keyword } = req.body; const file = req.file; if (!keyword || !file) { return res.status(400).json({ error: 'Thiếu keyword hoặc file' }); } const db = readDB(); db[keyword.toLowerCase()] = { fileName: file.originalname, storedName: file.filename, mimeType: file.mimetype, size: file.size }; writeDB(db); res.json({ success: true, message: 'Lưu file thành công!' }); }); // 3. API Lấy danh sách file (lệnh list) app.get('/api/files', (req, res) => { const db = readDB(); const list = Object.keys(db).map(kw => ({ keyword: kw, fileName: db[kw].fileName, size: db[kw].size })); res.json(list); }); // 4. API Download file theo từ khóa app.get('/api/download/:keyword', (req, res) => { const kw = req.params.keyword.toLowerCase(); const db = readDB(); const record = db[kw]; if (!record) { return res.status(404).json({ error: 'Không tìm thấy file' }); } const filePath = path.join(uploadDir, record.storedName); if (!fs.existsSync(filePath)) { return res.status(404).json({ error: 'File gốc đã bị xóa trên server' }); } res.download(filePath, record.fileName); }); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log(`Server chạy tại port ${PORT}`)); { "name": "grok-redblack-server", "version": "1.0.0", "main": "server.js", "scripts": { "start": "node server.js" }, "dependencies": { "cors": "^2.8.5", "express": "^4.18.2", "multer": "^1.4.5-lts.1" } }
